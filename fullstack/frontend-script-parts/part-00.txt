import { mkdir, readFile, readdir, writeFile } from 'node:fs/promises';
import { gunzipSync } from 'node:zlib';
import path from 'node:path';

async function readSorted(directory) {
  const names = (await readdir(directory)).filter((name) => name.endsWith('.txt')).sort();
  if (!names.length) throw new Error(`No source parts found in ${directory}`);
  return Promise.all(names.map((name) => readFile(path.join(directory, name), 'utf8')));
}

async function assembleWorker() {
  const parts = await readSorted('src-parts');
  await mkdir('src', { recursive: true });
  await writeFile('src/index.js', parts.join(''), 'utf8');
}

function replaceLine(lines, prefix, replacement) {
  const index = lines.findIndex((line) => line.startsWith(prefix));
  if (index < 0) throw new Error(`Unable to find source line: ${prefix}`);
  lines.splice(index, 1, ...replacement.split('\n'));
}
