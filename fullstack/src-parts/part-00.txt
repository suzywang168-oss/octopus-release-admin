function json(data, status = 200, headers = {}) {
  return new Response(JSON.stringify(data), {
    status,
    headers: { ...JSON_HEADERS, ...headers },
  });
}

function httpError(status, message) {
  const error = new Error(message);
  error.status = status;
  return error;
}

function nowIso() {
  return new Date().toISOString();
}

function randomId(prefix = "") {
  return `${prefix}${prefix ? "_" : ""}${crypto.randomUUID().replaceAll("-", "")}`;
}

function base64Url(bytes) {
  let binary = "";
  for (const byte of bytes) binary += String.fromCharCode(byte);
  return btoa(binary).replaceAll("+", "-").replaceAll("/", "_").replaceAll("=", "");
}

function fromBase64Url(value) {
  const normalized = value.replaceAll("-", "+").replaceAll("_", "/");
  const padded = normalized + "=".repeat((4 - (normalized.length % 4)) % 4);
  const binary = atob(padded);
  return Uint8Array.from(binary, (char) => char.charCodeAt(0));
}

function randomToken(size = 32) {
  const bytes = new Uint8Array(size);
  crypto.getRandomValues(bytes);
  return base64Url(bytes);
}

async function sha256(value) {
  const data = typeof value === "string" ? encoder.encode(value) : value;
  const digest = await crypto.subtle.digest("SHA-256", data);
  return base64Url(new Uint8Array(digest));
}

async function derivePassword(password, salt, iterations = 210000) {
  const keyMaterial = await crypto.subtle.importKey(
    "raw",
    encoder.encode(password),
    "PBKDF2",
    false,
    ["deriveBits"],
  );
  const bits = await crypto.subtle.deriveBits(
    { name: "PBKDF2", hash: "SHA-256", salt, iterations },
    keyMaterial,
    256,
  );
  return base64Url(new Uint8Array(bits));
}

async function createPassword(password, iterations = 210000) {
  const salt = new Uint8Array(16);
  crypto.getRandomValues(salt);
  return {
    hash: await derivePassword(password, salt, iterations),
    salt: base64Url(salt),
    iterations,
  };
}

async function verifyPassword(password, user) {
  const calculated = await derivePassword(
    password,
    fromBase64Url(user.password_salt),
    Number(user.password_iterations),
  );
  return timingSafeEqual(calculated, user.password_hash);
}

function timingSafeEqual(left, right) {
  if (left.length !== right.length) return false;
  let mismatch = 0;
  for (let index = 0; index < left.length; index += 1) {
    mismatch |= left.charCodeAt(index) ^ right.charCodeAt(index);
  }
  return mismatch === 0;
}

function parseCookies(request) {
  const raw = request.headers.get("cookie") || "";
  return Object.fromEntries(
    raw
      .split(";")
      .map((part) => part.trim())
      .filter(Boolean)
      .map((part) => {
        const index = part.indexOf("=");
        return index < 0
          ? [decodeURIComponent(part), ""]
          : [decodeURIComponent(part.slice(0, index)), decodeURIComponent(part.slice(index + 1))];
      }),
  );
}

function sessionCookie(token, maxAgeSeconds) {
  return [
    `octopus_session=${encodeURIComponent(token)}`,
    "Path=/",
    "HttpOnly",
    "Secure",
    "SameSite=Lax",
    `Max-Age=${maxAgeSeconds}`,
  ].join("; ");
}

function clearSessionCookie() {
  return "octopus_session=; Path=/; HttpOnly; Secure; SameSite=Lax; Max-Age=0";
}

async function readBody(request) {
  const contentLength = Number(request.headers.get("content-length") || 0);
  if (contentLength > 1_000_000) throw httpError(413, "请求数据过大");
  const text = await request.text();
  if (text.length > 1_000_000) throw httpError(413, "请求数据过大");
  if (!text) return {};
  try {
    return JSON.parse(text);
  } catch {
    throw httpError(400, "JSON 格式无效");
  }
}

function clientIp(request) {
  return request.headers.get("cf-connecting-ip") || "unknown";
}

function assertSameOrigin(request) {
  const origin = request.headers.get("origin");
  if (origin && origin !== new URL(request.url).origin) {
    throw httpError(403, "跨站请求已被阻止");
  }
}

async function rateLimit(env, key, limit, windowSeconds) {
  const windowStart = Math.floor(Date.now() / 1000 / windowSeconds) * windowSeconds;
  const current = await env.DB.prepare(
    "SELECT count, window_start FROM rate_limits WHERE key = ?",
  ).bind(key).first();
  if (!current || Number(current.window_start) !== windowStart) {
    await env.DB.prepare(
      "INSERT INTO rate_limits(key, count, window_start) VALUES(?, 1, ?) ON CONFLICT(key) DO UPDATE SET count = 1, window_start = excluded.window_start",
    ).bind(key, windowStart).run();
    return;
  }
  if (Number(current.count) >= limit) throw httpError(429, "请求过于频繁，请稍后再试");
  await env.DB.prepare("UPDATE rate_limits SET count = count + 1 WHERE key = ?").bind(key).run();
}
